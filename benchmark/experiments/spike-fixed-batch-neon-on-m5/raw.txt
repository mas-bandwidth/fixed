# suite.sh -- stand-in for `cmake --build` + `ctest` on a host with no cmake. Compiles each
# CMake test target with the flags a Release build uses (-O3 -DNDEBUG -std=c11
# -ffp-contract=off) and the same defines/sources/arguments CMakeLists.txt gives it, runs it,
# and counts. WILL_FAIL targets pass only when they exit non-zero.

CF="-O3 -DNDEBUG -std=c11 -ffp-contract=off -Iinclude"

pass=0; fail=0; failed=""
run() { # name willfail defs sources -- args
  name=$1; wf=$2; defs=$3; srcs=$4; shift 4
  clang $CF $defs $srcs -lm -o "$OUT/$name" 2>"$OUT/$name.cc" || { echo "BUILD FAIL $name"; cat "$OUT/$name.cc"|head -20; fail=$((fail+1)); failed="$failed $name"; return; }
  "$OUT/$name" "$@" >"$OUT/$name.log" 2>&1; rc=$?
  if [ "$wf" = 1 ]; then [ $rc -ne 0 ] && ok=1 || ok=0; else [ $rc -eq 0 ] && ok=1 || ok=0; fi
  if [ $ok = 1 ]; then pass=$((pass+1)); echo "PASS $name (rc=$rc wf=$wf)"; else fail=$((fail+1)); failed="$failed $name"; echo "FAIL $name (rc=$rc wf=$wf)"; tail -5 "$OUT/$name.log"; fi
}

== SOURCE OF sweep.sh (interleaved A/B + A/A driver; ab.sh minus its oracle step) ==
set -u
cd "$(dirname "$0")/.."
# ab.sh minus its oracle step (the oracle is run separately, once, by the real ab.sh and by suite.sh)
sed '/^# The candidate must also be bit-identical/,/^run_side "\$ISA_B" "\$WORK\/oracle_check"/d' benchmark/ab.sh > b/ab_nooracle.sh
export AB_REPEATS=${AB_REPEATS:-15} AB_ITERS=${AB_ITERS:-20000000} CC=clang
abrun() { # retry while ab.sh's own pgrep sees a transient match (it never kills; neither do we)
  n=0
  while :; do
    out=$(sh b/ab_nooracle.sh . . fixMulBatch fixDotBatch3 fixLengthBatch fixCrossBatch 2>&1); rc=$?
    if [ $rc -ne 3 ]; then echo "$out"; return; fi
    n=$((n+1)); echo "(ab.sh refused rc=3, attempt $n; holder: $(ps -Ao pid,command | grep -E 'op_bench|numerical_bench|oracle_check' | grep -v grep | head -3))"
    [ $n -ge 20 ] && { echo "$out"; return; }; sleep 3
  done
}
echo "##### sweep start $(date -u +%FT%TZ) loadavg: $(sysctl -n vm.loadavg)"
for cfg in "natural:0:0" "mulingate:33:0" "outofgate:0:14"; do
  name=${cfg%%:*}; rest=${cfg#*:}; sh_=${rest%%:*}; lsh=${rest#*:}
  for w in 16 256 4096; do
    export OP_BENCH_LANES=$w OP_BENCH_SHIFT=$sh_ OP_BENCH_LSHIFT=$lsh
    echo "##### config=$name lanes=$w rshift=$sh_ lshift=$lsh  A/B scalar vs neon"
    AB_ISA_A=scalar AB_ISA_B=neon abrun
    echo "##### config=$name lanes=$w rshift=$sh_ lshift=$lsh  A/A scalar vs scalar"
    AB_ISA_A=scalar AB_ISA_B=scalar abrun
    echo "##### loadavg after: $(sysctl -n vm.loadavg)"
  done
done
echo "##### sweep end $(date -u +%FT%TZ)"
